local function getSize(obj)
    if obj:IsA("BasePart") then return obj.Size end
    local ok,size=pcall(function() return obj:GetExtentsSize() end)
    if ok and size then return size end
    local INF=math.huge
    local x0,y0,z0,x1,y1,z1=INF,INF,INF,-INF,-INF,-INF
    for _,v in ipairs(obj:GetDescendants()) do
        if v:IsA("BasePart") then
            local cf,h=v.CFrame,v.Size*.5
            for _,sx in ipairs({-1,1}) do for _,sy in ipairs({-1,1}) do for _,sz in ipairs({-1,1}) do
                local p=cf:PointToWorldSpace(Vector3.new(sx*h.X,sy*h.Y,sz*h.Z))
                x0,y0,z0=math.min(x0,p.X),math.min(y0,p.Y),math.min(z0,p.Z)
                x1,y1,z1=math.max(x1,p.X),math.max(y1,p.Y),math.max(z1,p.Z)
            end end end
        end
    end
    return Vector3.new(x1-x0,y1-y0,z1-z0)
end

local function getAABBCenter(obj)
    if obj:IsA("BasePart") then return obj.Position end
    local INF=math.huge
    local x0,y0,z0,x1,y1,z1=INF,INF,INF,-INF,-INF,-INF
    for _,v in ipairs(obj:GetDescendants()) do
        if v:IsA("BasePart") then
            local cf,h=v.CFrame,v.Size*.5
            for _,sx in ipairs({-1,1}) do for _,sy in ipairs({-1,1}) do for _,sz in ipairs({-1,1}) do
                local p=cf:PointToWorldSpace(Vector3.new(sx*h.X,sy*h.Y,sz*h.Z))
                x0,y0,z0=math.min(x0,p.X),math.min(y0,p.Y),math.min(z0,p.Z)
                x1,y1,z1=math.max(x1,p.X),math.max(y1,p.Y),math.max(z1,p.Z)
            end end end
        end
    end
    return Vector3.new((x0+x1)*.5,(y0+y1)*.5,(z0+z1)*.5)
end

local function scaleToMatch(newObj,oldObj)
    local oldSize,newSize=getSize(oldObj),getSize(newObj)
    if newSize.X<.001 or newSize.Y<.001 or newSize.Z<.001 then return end
    local sx,sy,sz=oldSize.X/newSize.X,oldSize.Y/newSize.Y,oldSize.Z/newSize.Z
    if newObj:IsA("BasePart") then newObj.Size=oldSize; return end
    if not (newObj:IsA("Model") or newObj:IsA("Folder")) then return end
    if newObj:IsA("Model") then
        local geoMean=(sx*sy*sz)^(1/3)
        local ok=pcall(function() newObj:ScaleTo(geoMean) end)
        if ok then
            local after=getSize(newObj)
            local eX=math.abs(after.X-oldSize.X)/math.max(oldSize.X,.001)
            local eY=math.abs(after.Y-oldSize.Y)/math.max(oldSize.Y,.001)
            local eZ=math.abs(after.Z-oldSize.Z)/math.max(oldSize.Z,.001)
            if eX<=.12 and eY<=.12 and eZ<=.12 then return end
            sx=oldSize.X/math.max(after.X,.001); sy=oldSize.Y/math.max(after.Y,.001); sz=oldSize.Z/math.max(after.Z,.001)
        end
    end
    local okp,pivot=pcall(function() return newObj:GetPivot() end)
    if not okp then pivot=CFrame.new(getAABBCenter(newObj)) end
    for _,part in ipairs(newObj:GetDescendants()) do
        if part:IsA("BasePart") then
            local rel=pivot:PointToObjectSpace(part.Position)
            local newRel=Vector3.new(rel.X*sx,rel.Y*sy,rel.Z*sz)
            part.Size=Vector3.new(part.Size.X*sx,part.Size.Y*sy,part.Size.Z*sz)
            part.CFrame=CFrame.new(pivot:PointToWorldSpace(newRel))*(part.CFrame-part.CFrame.Position)
        end
    end
end

local function moveToPosition(newObj,targetCenter)
    local shift=targetCenter-getAABBCenter(newObj)
    if newObj:IsA("Model") then
        local curPivot=newObj:GetPivot()
        newObj:PivotTo(CFrame.new(curPivot.Position+shift)*(curPivot-curPivot.Position))
    elseif newObj:IsA("BasePart") then
        newObj.Position=newObj.Position+shift
    else
        for _,d in ipairs(newObj:GetDescendants()) do
            if d:IsA("BasePart") then d.Position=d.Position+shift end
        end
    end
end

local function resolveObjects(obj)
    local parent=obj.Parent
    if parent and parent~=workspace and (parent:IsA("Model") or parent:IsA("Folder")) then
        local pool={}
        for _,child in ipairs(parent:GetChildren()) do
            if child:IsA("Model") or child:IsA("BasePart") then table.insert(pool,child) end
        end
        if #pool>=1 then return pool,parent end
    end
    return {obj},nil
end

local function stopSelect()
    selectMode=nil
    if selectConn then selectConn:Disconnect(); selectConn=nil end
    srcBtn.Text="SELECT"; tgtBtn.Text="SELECT"
end

local function startSelect(mode)
    stopSelect(); selectMode=mode
    local btn=(mode=="source") and srcBtn or tgtBtn
    btn.Text="CANCEL"; setStatus("Click an object in 3D world...",COLORS.WARNING)
    selectConn=Mouse.Button1Down:Connect(function()
        local target=Mouse.Target
        if not target then setStatus("Missed. Click a valid part.",COLORS.DANGER); return end
        local obj=target
        if obj:IsA("BasePart") and obj.Parent and obj.Parent:IsA("Model") then obj=obj.Parent end
        local pool,folder=resolveObjects(obj)
        if mode=="source" then
            srcList=pool
            if folder then srcNameLbl.Text="📂 "..folder.Name; setStatus("Source folder loaded.",COLORS.SUCCESS)
            else srcNameLbl.Text="📦 "..obj.Name; setStatus("Source object loaded.",COLORS.SUCCESS) end
        else
            tgtPool=pool
            if folder then tgtNameLbl.Text="📂 "..folder.Name; setStatus("Target pool loaded.",COLORS.SUCCESS)
            else tgtNameLbl.Text="📦 "..obj.Name; setStatus("Target object loaded.",COLORS.SUCCESS) end
        end
        updateUI(); stopSelect()
    end)
end

ReplaceBtn.MouseButton1Click:Connect(function()
    if #srcList==0 or #tgtPool==0 then setStatus("Select Source & Target first!",COLORS.DANGER); return end
    ReplaceBtn.Text="PROCESSING..."; tween(ReplaceBtn,{BackgroundColor3=COLORS.WARNING})
    local tgtSet,toReplace={},{}
    for _,t in ipairs(tgtPool) do tgtSet[t]=true end
    for _,s in ipairs(srcList) do if s and s.Parent and not tgtSet[s] then table.insert(toReplace,s) end end
    if #toReplace==0 then
        setStatus("Source objects no longer exist.",COLORS.DANGER)
        ReplaceBtn.Text="REPLACE 0 OBJECTS"; tween(ReplaceBtn,{BackgroundColor3=COLORS.ACCENT}); return
    end
    task.spawn(function()
        local ok,fail=0,0
        for i,old in ipairs(toReplace) do
            local success=pcall(function()
                local chosen=tgtPool[math.random(1,#tgtPool)]
                local oldCenter=getAABBCenter(old)
                local newClone=chosen:Clone(); newClone.Name=chosen.Name
                scaleToMatch(newClone,old); newClone.Parent=old.Parent
                moveToPosition(newClone,oldCenter); old:Destroy()
            end)
            if success then ok=ok+1 else fail=fail+1 end
            if i%10==0 then RunService.Heartbeat:Wait() end
        end
        tween(ReplaceBtn,{BackgroundColor3=COLORS.ACCENT})
        ReplaceBtn.Text="REPLACE "..tostring(#srcList).." OBJECTS"
        if fail==0 then setStatus("Success! "..ok.." items replaced.",COLORS.SUCCESS)
        else setStatus("Done: "..ok.." OK, "..fail.." FAILED.",COLORS.WARNING) end
        srcList={}; srcNameLbl.Text="None Selected"; updateUI()
    end)
end)
